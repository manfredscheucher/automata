// Input: a binary string w
// Output: accepts if w is a palindrome
// Example: accepts 11011
// Simulator: https://turingmachinesimulator.com (2 tapes)
//
// Copies the input to tape 2, moves the head of tape 1 back to the start,
// then compares: tape 1 from left to right, tape 2 from right to left.
// About 3n steps instead of about n^2/2 with one tape.

name: Binary palindrome with 2 tapes
init: copy
accept: done

// Copy the input to tape 2.
copy,0,_
copy,0,0,>,>

copy,1,_
copy,1,1,>,>

copy,_,_
rewind,_,_,<,<

// Move the head of tape 1 back to the start, tape 2 stays at its end.
rewind,0,0
rewind,0,0,<,-

rewind,0,1
rewind,0,1,<,-

rewind,0,_
rewind,0,_,<,-

rewind,1,0
rewind,1,0,<,-

rewind,1,1
rewind,1,1,<,-

rewind,1,_
rewind,1,_,<,-

rewind,_,0
compare,_,0,>,-

rewind,_,1
compare,_,1,>,-

rewind,_,_
compare,_,_,>,-

// Compare: tape 1 goes right, tape 2 goes left. A mismatch has no rule, so it rejects.
compare,0,0
compare,0,0,>,<

compare,1,1
compare,1,1,>,<

compare,_,_
done,_,_,-,-
