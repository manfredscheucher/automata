// copy 1s
// Ported from the turingmachine.io example copy1s by Andy Li (BSD 3-Clause):
// https://github.com/aepsilon/turing-machine-viz/blob/master/src/examples/copy1s.yaml
// The blank '0' is written as _ here.
// Example input: 111
//
// Copies a string of consecutive 1s.

name: copy 1s
init: each
accept: H

// mark the current 1 by erasing it
each,_
H,_,>

each,1
sep,_,>

// skip to the separator
sep,_
add,_,>

sep,1
sep,1,>

// skip to the end of the copy and write a 1
add,_
sepL,1,<

add,1
add,1,>

// return to the separator
sepL,_
next,_,<

sepL,1
sepL,1,<

// return to the erased 1, restore it, and then advance to the next 1
next,_
each,1,>

next,1
next,1,<
