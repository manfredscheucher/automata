// three equal lengths
// Ported from the turingmachine.io example matchThreeLengths by Andy Li (BSD 3-Clause license):
// https://github.com/aepsilon/turing-machine-viz/blob/master/src/examples/matchThreeLengths.yaml
// The blank ' ' is written as _ here.
// Example input: aabbcc (try bac, aabc, aabcc, aabcbc)
//
// Decides the language { aⁿbⁿcⁿ | n ≥ 1 }, that is,
// accepts a's followed by b's then c's of the same length.
// Mark the first a, b, and c on each pass (by capitalizing them).
// All a's must precede all b's, which must precede all c's.
// When there are no more a's,
// all input symbols should have been marked.

name: three equal lengths
init: qA
accept: accept

qA,a
qB,A,>

qA,B
scan,B,>

qB,a
qB,a,>

qB,B
qB,B,>

qB,b
qC,B,>

qC,b
qC,b,>

qC,C
qC,C,>

qC,c
back,C,<

back,a
back,a,<

back,B
back,B,<

back,b
back,b,<

back,C
back,C,<

back,A
qA,A,>

scan,B
scan,B,>

scan,C
scan,C,>

scan,_
accept,_,>
