// unary multiplication
// Ported from the turingmachine.io example unaryMult by Andy Li (BSD 3-Clause):
// https://github.com/aepsilon/turing-machine-viz/blob/master/src/examples/unaryMult.yaml
// The blank ' ' is written as _ here.
// Example input: ||*||| (try '*', '|*|||', '||||*||')
//
// Multiplies together two unary numbers separated by a '*'.
// (Unary is like tallying. Here '||*|||' means 2 times 3.)
//
// The idea:
//   multiply(0, b) = 0
//   multiply(a, b) = b + multiply(a-1, b)   when a > 0

name: unary multiplication
init: eachA
accept: done

// For each 1 in a, add a copy of b.
// Inductive case: a > 0.
eachA,|
toB,_,>

// Base case:      a = 0.
eachA,*
skip,*,>

toB,|
toB,|,>

// enter copier
toB,*
eachB,*,>

// return from copier
nextA,_
eachA,|,>

nextA,|
nextA,|,<

nextA,*
nextA,*,<

skip,|
skip,|,>

skip,_
done,_,>

// This is the 'copy 1s' machine, with ' ' as 0 and '|' as 1.
eachB,_
nextA,_,<

eachB,|
sep,_,>

sep,_
add,_,>

sep,|
sep,|,>

add,_
sepL,|,<

add,|
add,|,>

sepL,_
nextB,_,<

sepL,|
sepL,|,<

nextB,_
eachB,|,>

nextB,|
nextB,|,<
