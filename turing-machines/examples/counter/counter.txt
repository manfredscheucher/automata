// Input: nothing (the tape starts with 0) or a binary number n
// Output: never stops, the tape counts 0, 1, 10, 11, ... (or n, n+1, n+2, ...)
// Simulator: https://turingmachinesimulator.com
//
// The increment from increment.txt in a loop: after each +1 it goes
// to the rightmost digit again instead of stopping.
//
// Based on the turingmachine.io example binaryIncrement:
// https://github.com/aepsilon/turing-machine-viz/blob/master/src/examples/binaryIncrement.yaml
// Copyright 2015-2021 Andy Li, BSD 3-Clause license, see ../turingmachine-io/LICENSE.

name: Binary counter
init: start
accept: none

// On a blank tape write 0 first. Otherwise start with the input.
start,_
right,0,-

start,0
right,0,>

start,1
right,1,>

// Go to the rightmost digit.
right,0
right,0,>

right,1
right,1,>

right,_
carry,_,<

// Add 1: 1 + carry = 0, carry 1 to the next digit.
carry,1
carry,0,<

// 0 + carry = 1, then start over with the next +1.
carry,0
right,1,>

carry,_
right,1,>
